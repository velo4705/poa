cmake_minimum_required(VERSION 3.16)
project(epson-iprojection-linux VERSION 0.1 LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED gtk4)
find_package(OpenSSL REQUIRED)
add_executable(epson-installer main.c)
target_include_directories(epson-installer PRIVATE ${GTK4_INCLUDE_DIRS})
target_link_directories(epson-installer PRIVATE ${GTK4_LIBRARY_DIRS})
target_link_libraries(epson-installer PRIVATE ${GTK4_LIBRARIES} OpenSSL::Crypto)
